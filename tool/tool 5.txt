const docProps = PropertiesService.getDocumentProperties();
const WP_URL = docProps.getProperty('WP_URL') || "https://binhanvosu.xo.je"; 
const WP_USER = docProps.getProperty('WP_USER') || "sa";
const WP_APP_PASSWORD = docProps.getProperty('WP_PASS') || "xxxx xxxx xxxx xxxx"; 
const WP_CAT_RAW = docProps.getProperty('WP_CAT_MAP') || "Mặc định: 1"; 

const IMAGE_EXTENSION = ".webp"; 

function getCurrentWpImageFolder() {
  const date = new Date();
  const year = date.getFullYear(); 
  const month = String(date.getMonth() + 1).padStart(2, '0'); 
  return `${WP_URL}/wp-content/uploads/${year}/${month}`;
}

const WP_IMAGE_FOLDER = getCurrentWpImageFolder();

// 1. HÀM CHÍNH (MASTER) - ĐÃ CÁCH LY HOÀN TOÀN KHỎI TOOL 7
// -------------------------------------------------------------
function masterPostProcessTool5() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();

  const CATEGORY_MAP = { "Mặc định": 1 };
  let catPairs = WP_CAT_RAW.split(",");
  for (let c of catPairs) {
    let parts = c.split(":");
    if (parts.length === 2) {
      let catName = parts[0].trim();
      let catId = parts[1].trim();
      if (catName && catId) CATEGORY_MAP[catName] = catId;
    }
  }

  for (let i = 0; i < data.length; i++) {
    let rawKey        = String(data[i][0] || "").trim();   
    let docUrl        = String(data[i][1] || "").trim();   
    let categoryName  = String(data[i][2] || "").trim();   
    let postStatusRaw = String(data[i][3] || "").trim();   
    let runStatus     = String(data[i][4] || "").trim();   
    
    if(!rawKey || !docUrl.includes("docs.google.com")) continue;
    if(runStatus.includes("✅")) continue;

    let categoryId = CATEGORY_MAP[categoryName] || CATEGORY_MAP["Mặc định"] || 1;
    let finalPostStatus = (postStatusRaw.toLowerCase() === "public" || postStatusRaw.toLowerCase() === "publish") ? "publish" : "draft";

    try {
      const docIdMatch = docUrl.match(/[-\w]{25,}/);
      if (!docIdMatch) {
         sheet.getRange(i + 1, 5).setValue("❌ Lỗi URL Google Docs");
         continue;
      }
      
      const postSlug = convertToPostSlugTool5(rawKey);              
      const imageSlug = convertToImageSlugTool5(rawKey);     

      const finalData = buildArticleContentTool5(docIdMatch[0], imageSlug);

      const thumbData = getMediaDataByFilenameTool5(imageSlug + IMAGE_EXTENSION);
      const thumbnailId = thumbData ? thumbData.id : null;

      if(!thumbnailId) {
        let generatedThumbUrl = `${WP_IMAGE_FOLDER}/${imageSlug}${IMAGE_EXTENSION}`;
        finalData.content = `<p style="text-align: center;"><img src="${generatedThumbUrl}" alt="${rawKey.replace(/"/g, '&quot;')}" class="size-full" /></p>\n` + finalData.content;
      } else {
        updateMediaMetadataTool5(thumbnailId, rawKey);
      }
      
      const responseWP = sendFullPostToWPTool5(finalData, thumbnailId, postSlug, rawKey, categoryId, finalPostStatus);

      if (responseWP && responseWP.success) {
        sheet.getRange(i + 1, 5).setValue("✅"); 
        if (responseWP.postLink) sheet.getRange(i + 1, 6).setValue(responseWP.postLink); 
      } else {
        sheet.getRange(i + 1, 5).setValue("❌ Lỗi: " + (responseWP ? responseWP.errorMsg : "WP Tịt"));
      }
      Utilities.sleep(3000);
    } catch (e) {
      sheet.getRange(i + 1, 5).setValue("❌ Lỗi Code: " + e.message);
    }
  }
}

// 2A. TẠO SLUG LINK BÀI (Dùng -)
// -------------------------------------------------------------
function convertToPostSlugTool5(str) {
  if (!str) return "";
  str = String(str).toLowerCase().trim();
  str = str.replace(/à|á|ạ|ả|ã|â|ầ|ấ|ậ|ẩ|ẫ|ă|ằ|ắ|ặ|ẳ|ẵ/g, "a");
  str = str.replace(/è|é|ẹ|ẻ|ẽ|ê|ề|ế|ệ|ể|ễ/g, "e");
  str = str.replace(/ì|í|ị|ỉ|ĩ/g, "i");
  str = str.replace(/ò|ó|ọ|ỏ|õ|ô|ồ|ố|ộ|ổ|ỗ|ơ|ờ|ớ|ợ|ở|ỡ/g, "o");
  str = str.replace(/ù|ú|ụ|ủ|ũ|ư|ừ|ứ|ự|ử|ữ/g, "u");
  str = str.replace(/ỳ|ý|ỵ|ỷ|ỹ/g, "y");
  str = str.replace(/đ/g, "d");
  str = str.replace(/\//g, "-"); 
  str = str.replace(/[^a-z0-9 -]/g, ""); 
  str = str.replace(/\s+/g, "-");        
  str = str.replace(/-+/g, "-");         
  return str.replace(/^-+|-+$/g, ""); 
}

// 3. HÀM TÌM ẢNH (ÉP KHỚP TUYỆT ĐỐI 100% - CHỐNG LẤY NHẦM THUMB)
// -------------------------------------------------------------
function getMediaDataByFilenameTool5(filename) {
  let cleanName = filename.replace(/\.(webp|jpg|png|jpeg)$/i, "");
  let wpSlug = cleanName.replace(/_/g, "-"); 

  const options = {
    method: "GET",
    headers: { "Authorization": "Basic " + Utilities.base64Encode(WP_USER + ":" + WP_APP_PASSWORD) },
    muteHttpExceptions: true
  };

  const resSlug = UrlFetchApp.fetch(WP_URL + "/wp-json/wp/v2/media?slug=" + encodeURIComponent(wpSlug), options);
  if (resSlug.getResponseCode() === 200) {
    const data = JSON.parse(resSlug.getContentText());
    if (data.length > 0) {
      for (let d of data) {
        if (d.source_url) {
          let wpFileName = d.source_url.split('/').pop().replace(/\.(webp|jpg|png|jpeg)$/i, "");
          // Chỉ lấy nếu giống nhau y đúc (===)
          if (wpFileName === cleanName) {
            return { id: d.id, url: d.source_url, w: d.media_details.width, h: d.media_details.height };
          }
        }
      }
    }
  }

  const resSearch = UrlFetchApp.fetch(WP_URL + "/wp-json/wp/v2/media?search=" + encodeURIComponent(cleanName), options);
  if (resSearch.getResponseCode() === 200) {
    const dataSearch = JSON.parse(resSearch.getContentText());
    for(let d of dataSearch) {
      if (d.source_url) {
        let wpFileName = d.source_url.split('/').pop().replace(/\.(webp|jpg|png|jpeg)$/i, "");
        // Chỉ lấy nếu giống nhau y đúc (===)
        if (wpFileName === cleanName) {
          return { id: d.id, url: d.source_url, w: d.media_details.width, h: d.media_details.height };
        }
      }
    }
  }
  return null; 
}

// 4. HÀM RÁP NỘI DUNG (CÓ MỚI GẮP - KHÔNG CÓ THÌ BỎ QUA)
// -------------------------------------------------------------
function buildArticleContentTool5(docId, imageSlug) {
  const doc = DocumentApp.openById(docId);
  const body = doc.getBody();
  const numChildren = body.getNumChildren();
  let metaDesc = "", h1Title = "", contentStartIndex = 3; 

  let textBlocks = [];
  for (let i = 0; i < Math.min(8, numChildren); i++) {
    const child = body.getChild(i);
    if (child.getType() === DocumentApp.ElementType.PARAGRAPH) {
      let t = child.asText().getText().trim();
      if (t.length > 2 && !t.includes("drive.google")) textBlocks.push({ index: i, text: t });
    }
  }
  if (textBlocks.length >= 2) { metaDesc = textBlocks[0].text; h1Title = textBlocks[1].text; contentStartIndex = textBlocks[1].index + 1; }

  let htmlBody = "", imageCounter = 1, previousWasHeading = false, inList = false, currentListType = "";

  for (let i = contentStartIndex; i < numChildren; i++) {
    const child = body.getChild(i);
    const type = child.getType();

    if (type === DocumentApp.ElementType.LIST_ITEM) {
      let listItem = child.asListItem();
      let listType = (listItem.getGlyphType() === DocumentApp.GlyphType.NUMBER) ? "ol" : "ul";
      if (!inList) { htmlBody += `<${listType} style="text-align: justify;">\n`; inList = true; currentListType = listType; }
      htmlBody += "  <li>" + extractFormattedTextTool5(listItem) + "</li>\n";
      previousWasHeading = false; 
    } else if (type === DocumentApp.ElementType.PARAGRAPH) {
      if (inList) { htmlBody += `</${currentListType}>\n`; inList = false; }
      let p = child.asParagraph();
      let text = p.getText().trim();
      if (text.length === 0) continue; 

      let isHeading = p.getHeading() !== DocumentApp.ParagraphHeading.NORMAL;
      let isCaption = !isHeading && text.length < 150 && previousWasHeading;

      if (isCaption) {
        let targetFileName = `${imageSlug}-${imageCounter}${IMAGE_EXTENSION}`;
        let mediaData = getMediaDataByFilenameTool5(targetFileName);
        
        let cleanText = text.replace(/[\r\n\v]+/g, " ").trim(); 
        let safeAlt = cleanText.replace(/"/g, '&quot;'); 

        if (mediaData && mediaData.id) {
          updateMediaMetadataTool5(mediaData.id, cleanText);
          
          let captionWidth = mediaData.w ? mediaData.w : 1200; 
          let imgWidth = mediaData.w ? ` width="${mediaData.w}"` : ` width="${captionWidth}"`;
          let imgHeight = mediaData.h ? ` height="${mediaData.h}"` : "";
          
          htmlBody += `\n[caption id="attachment_${mediaData.id}" align="aligncenter" width="${captionWidth}"]<img class="wp-image-${mediaData.id} size-full" src="${mediaData.url}" alt="${safeAlt}"${imgWidth}${imgHeight} /> ${cleanText}[/caption]\n\n`;
          imageCounter++; 
        } else {
          // Bỏ qua hoàn toàn việc tạo link ảo nếu WordPress báo Không tìm thấy ảnh
          htmlBody += `<p style="text-align: justify;">${extractFormattedTextTool5(p)}</p>\n`;
        }
        
        previousWasHeading = false; 
      } else {
        htmlBody += convertToHtmlTool5(p) + "\n";
        previousWasHeading = isHeading;
      }
    }
  }
  return { title: h1Title, content: htmlBody, meta: metaDesc };
}

function convertToHtmlTool5(paragraph) {
    const textStr = paragraph.getText().trim();
    if (textStr === "") return ""; 
    const heading = paragraph.getHeading();
    let tag = "p";
    if (heading === DocumentApp.ParagraphHeading.HEADING1 || heading === DocumentApp.ParagraphHeading.HEADING2) tag = "h2";
    if (heading === DocumentApp.ParagraphHeading.HEADING3) tag = "h3";
    return `<${tag} style="text-align: justify;">${extractFormattedTextTool5(paragraph)}</${tag}>`;
}

function extractFormattedTextTool5(element) {
    const textStr = element.getText().trim();
    const textObj = element.editAsText();
    const indices = textObj.getTextAttributeIndices(); 
    let innerHtml = "";
    for (let i = 0; i < indices.length; i++) {
        let start = indices[i], end = (i + 1 < indices.length) ? indices[i + 1] : textStr.length;
        if(start >= end) continue;
        let segment = textStr.substring(start, end).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        if (textObj.isBold(start)) segment = `<strong>${segment}</strong>`;
        if (textObj.isItalic(start)) segment = `<em>${segment}</em>`;
        let url = textObj.getLinkUrl(start);
        if (url) segment = `<a href="${url}" target="_blank" rel="noopener">${segment}</a>`;
        innerHtml += segment;
    }
    return innerHtml;
}

function sendFullPostToWPTool5(data, thumbId, slug, rawKey, catId, status) {
  const url = WP_URL + "/wp-json/wp/v2/posts";
  let payload = { title: data.title, content: data.content, slug: slug, status: status, categories: [catId], excerpt: data.meta };
  if (thumbId) payload.featured_media = thumbId;
  const options = { method: "POST", headers: { "Authorization": "Basic " + Utilities.base64Encode(WP_USER + ":" + WP_APP_PASSWORD), "Content-Type": "application/json" }, payload: JSON.stringify(payload), muteHttpExceptions: true };
  const res = UrlFetchApp.fetch(url, options);
  if (res.getResponseCode() === 201 || res.getResponseCode() === 200) return { success: true, postLink: JSON.parse(res.getContentText()).link };
  return { success: false, errorMsg: JSON.parse(res.getContentText()).message };
}

// 7. HÀM CẬP NHẬT SEO ẢNH (GIỮ NGUYÊN TIÊU ĐỀ GỐC CỦA WP)
// -------------------------------------------------------------
function updateMediaMetadataTool5(id, text) {
  if (!id) return;
  const url = WP_URL + "/wp-json/wp/v2/media/" + id;
  
  // Làm sạch văn bản, khử các ký tự xuống dòng gây lỗi mã HTML
  let cleanText = text.replace(/[\r\n\v]+/g, " ").trim(); 
  
  // Chỉ gửi 3 thông số cần thiết, bỏ qua trường 'title'
  const payload = {
    alt_text: cleanText,    // 1. Ô Văn bản thay thế (Mô tả SEO)
    caption: cleanText,     // 2. Ô Chú thích (Hiển thị dưới ảnh)
    description: cleanText  // 3. Ô Miêu tả (Dữ liệu bổ trợ SEO)
  };

  const options = { 
    method: "POST", 
    headers: { 
      "Authorization": "Basic " + Utilities.base64Encode(WP_USER + ":" + WP_APP_PASSWORD), 
      "Content-Type": "application/json" 
    }, 
    payload: JSON.stringify(payload), 
    muteHttpExceptions: true 
  };
  
  UrlFetchApp.fetch(url, options); 
}
